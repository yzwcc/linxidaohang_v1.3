cmake_minimum_required(VERSION 3.16)
project(lingxi_octomap LANGUAGES CXX)
find_package(OCTOMAP 1.9.7 EXACT REQUIRED CONFIG PATHS /usr/share/octomap NO_DEFAULT_PATH)
add_library(lingxi_octomap SHARED octomap_bridge.cpp)
target_include_directories(lingxi_octomap PRIVATE ${OCTOMAP_INCLUDE_DIRS})
target_link_libraries(lingxi_octomap PRIVATE ${OCTOMAP_LIBRARIES})
target_compile_features(lingxi_octomap PRIVATE cxx_std_17)
target_compile_options(lingxi_octomap PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(lingxi_octomap PRIVATE LINGXI_OCTOMAP_VERSION="${OCTOMAP_VERSION}")
