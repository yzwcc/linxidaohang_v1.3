cmake_minimum_required(VERSION 3.19)
project(lingxi_ground LANGUAGES CXX)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
if(NOT IS_DIRECTORY "${PATCHWORKPP_SOURCE}")
  message(FATAL_ERROR "Explicit pinned PATCHWORKPP_SOURCE required")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../patchworkpp.lock.json" LOCK_JSON)
string(JSON REVISION GET "${LOCK_JSON}" revision)
execute_process(COMMAND git -C "${PATCHWORKPP_SOURCE}" rev-parse HEAD
  OUTPUT_VARIABLE HEAD_REV OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE STATUS)
if(NOT STATUS EQUAL 0 OR NOT HEAD_REV STREQUAL REVISION)
  message(FATAL_ERROR "Pinned Patchwork++ commit required")
endif()
string(JSON COUNT LENGTH "${LOCK_JSON}" files)
math(EXPR LAST "${COUNT} - 1")
foreach(INDEX RANGE ${LAST})
  string(JSON NAME MEMBER "${LOCK_JSON}" files ${INDEX})
  string(JSON EXPECTED GET "${LOCK_JSON}" files "${NAME}")
  if(IS_SYMLINK "${PATCHWORKPP_SOURCE}/${NAME}")
    message(FATAL_ERROR "Direct upstream file required: ${NAME}")
  endif()
  file(SHA256 "${PATCHWORKPP_SOURCE}/${NAME}" ACTUAL)
  if(NOT ACTUAL STREQUAL EXPECTED)
    message(FATAL_ERROR "Changed upstream file: ${NAME}")
  endif()
endforeach()
add_library(lingxi_ground SHARED ground_bridge.cpp
  "${PATCHWORKPP_SOURCE}/cpp/patchworkpp/src/patchworkpp.cpp"
  "${PATCHWORKPP_SOURCE}/cpp/common/src/plane_fit.cpp")
target_include_directories(lingxi_ground PRIVATE
  "${PATCHWORKPP_SOURCE}/cpp/patchworkpp/include"
  "${PATCHWORKPP_SOURCE}/cpp/common/include")
target_link_libraries(lingxi_ground PRIVATE Eigen3::Eigen)
target_compile_features(lingxi_ground PRIVATE cxx_std_17)
target_compile_definitions(lingxi_ground PRIVATE LINGXI_PATCHWORKPP_REVISION="${REVISION}")
